#run on https://stepindev.com/en/py-playground
from turtle import *

def ichi(size):
    fd(size)
    rt(180)
    fd(size)
    lt(90)
    pu()
    fd(25)
    lt(90)
    pd()

def san(size):
    for i in range(0,3):
        ichi(size)

def square():
    pu()
    seth(45)
    fd(50)
    seth(0)
    pd()
    for i in range(0,4):
        rt(90)
        fd(50)
    pu()

def shina(size):
    pu()
    setpos(0,size)
    square()
    setpos(-size,-size)
    square()
    setpos(size,-size)
    square()

san(100)
resetscreen()
home()
shina(30)
